# text-layer backends above return ~0 chars, common/pdf_extract.py rasterizes

# `tesseract` system binary (apt: tesseract-ocr, brew: tesseract) and Pillow.
# Falls back gracefully (OCR skipped) when any of these are absent. Issue #856.
pytesseract>=0.3.10
Pillow>=9.0.0
